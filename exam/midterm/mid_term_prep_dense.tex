\mysection{OPTIMIZATION FUNDAMENTALS}

\mysubsection{Standard Forms}
\textbf{General}: $\min f(x)$ s.t. $g_i(x) \leq b_i$, $h_j(x) = c_j$, $x \in X$
\textbf{LP Standard}: $\min c^Tx$ s.t. $Ax = b$, $x \geq 0$
\textbf{LP Canonical}: $\min c^Tx$ s.t. $Ax \leq b$, $x \geq 0$

\mysubsection{Linear Algebra Essentials}
\textbf{Rank}: $\text{rank}(A) = $ \# linearly independent rows/cols
\textbf{Full rank}: $m \times n$ matrix, $\text{rank}(A) = \min(m,n)$
\textbf{Matrix inverse}: $AA^{-1} = I$ (exists iff $A$ square \& full rank)
\textbf{Determinant}: $\det(A) \neq 0 \Leftrightarrow A$ invertible
\textbf{Eigenvalues}: $Av = \lambda v$ where $\det(A - \lambda I) = 0$

\mysubsection{System Solutions $Ax = b$}
$m = n$, $\text{rank}(A) = n$: Unique solution $x = A^{-1}b$
$m < n$: Infinite solutions (underdetermined)
$m > n$, $\text{rank}(A) = n$: Least squares $x = (A^TA)^{-1}A^Tb$
$\text{rank}(A) < \text{rank}([A|b])$: No solution (inconsistent)

\mysubsection{Gradients \& Hessians}
$\nabla f = [\frac{\partial f}{\partial x_1}, \ldots, \frac{\partial f}{\partial x_n}]^T$
$\nabla^2 f = [\frac{\partial^2 f}{\partial x_i \partial x_j}]$ (Hessian matrix)
\textbf{Chain rule}: $\nabla_x f(g(x)) = \nabla g(x)^T \nabla_y f(g(x))$
\textbf{Product rule}: $\nabla(fg) = f\nabla g + g\nabla f$

\mysubsection{Taylor Expansions}
\textbf{1st order}: $f(x) \approx f(x_0) + \nabla f(x_0)^T(x-x_0)$
\textbf{2nd order}: $f(x) \approx f(x_0) + \nabla f(x_0)^T(x-x_0) + \frac{1}{2}(x-x_0)^T\nabla^2f(x_0)(x-x_0)$

\mysubsection{Optimality Conditions (Unconstrained)}
\textbf{1st order necessary}: $\nabla f(x^*) = 0$
\textbf{2nd order sufficient}:
- Min: $\nabla f(x^*) = 0$ \& $\nabla^2 f(x^*) \succ 0$ (PD)
- Max: $\nabla f(x^*) = 0$ \& $\nabla^2 f(x^*) \prec 0$ (ND)

\mysubsection{Matrix Definiteness}
\textbf{PD} ($A \succ 0$): $x^TAx > 0 \, \forall x \neq 0$, all $\lambda_i > 0$
\textbf{PSD} ($A \succeq 0$): $x^TAx \geq 0 \, \forall x$, all $\lambda_i \geq 0$
\textbf{ND} ($A \prec 0$): $x^TAx < 0 \, \forall x \neq 0$, all $\lambda_i < 0$
\textbf{NSD} ($A \preceq 0$): $x^TAx \leq 0 \, \forall x$, all $\lambda_i \leq 0$
\textbf{Indefinite}: Mixed signs in eigenvalues

\mysubsection{PSD Tests}
1. All eigenvalues $\geq 0$
2. All leading principal minors $\geq 0$
3. $x^TAx \geq 0$ for all $x$
4. $A = B^TB$ for some matrix $B$

\mysection{CONVEXITY THEORY}

\mysubsection{Convex Sets}
\textbf{Definition}: $S$ convex if $\lambda x + (1-\lambda)y \in S$ $\forall x,y \in S$, $\lambda \in [0,1]$

\textbf{Examples}:
- Hyperplane: $\{x | a^Tx = b\}$
- Halfspace: $\{x | a^Tx \leq b\}$
- Ball: $\{x | \|x-c\| \leq r\}$
- Ellipsoid: $\{x | (x-c)^TP^{-1}(x-c) \leq 1\}$, $P \succ 0$
- Polyhedron: $\{x | Ax \leq b\}$
- Simplex: $\{x | \sum x_i = 1, x_i \geq 0\}$

\textbf{Operations preserving convexity}:
- Intersection: $S_1 \cap S_2$ convex if $S_1, S_2$ convex
- Affine transformation: $\{Ax + b | x \in S\}$ convex if $S$ convex
- Cartesian product: $S_1 \times S_2$ convex if $S_1, S_2$ convex

\mysubsection{Convex Functions}
\textbf{Definition}: $f$ convex if $f(\lambda x + (1-\lambda)y) \leq \lambda f(x) + (1-\lambda)f(y)$

\textbf{Tests for Convexity}:
1. \textbf{Definition}: Check convex combination property
2. \textbf{1st order}: $f(y) \geq f(x) + \nabla f(x)^T(y-x)$ $\forall x,y$
3. \textbf{2nd order}: $\nabla^2 f(x) \succeq 0$ $\forall x$

\textbf{Common Convex Functions}:
- Linear: $a^Tx + b$
- Quadratic: $\frac{1}{2}x^TPx + q^Tx + r$ where $P \succeq 0$
- Exponential: $e^{ax}$, $a \in \mathbb{R}$
- Power: $x^p$ for $p \geq 1$, $x \geq 0$
- Logarithmic: $-\log(x)$ for $x > 0$
- Norms: $\|x\|_p$ for $p \geq 1$
- Max function: $\max\{f_1(x), f_2(x)\}$ if $f_i$ convex

\columnbreak

\mysubsection{Convexity Preservation}
- \textbf{Nonnegative combination}: $\sum \alpha_i f_i(x)$, $\alpha_i \geq 0$, $f_i$ convex
- \textbf{Pointwise maximum}: $\max_i f_i(x)$ if $f_i$ convex
- \textbf{Composition}: $g(f(x))$ convex if $g$ convex increasing, $f$ convex
- \textbf{Perspective}: $tf(x/t)$ convex if $f$ convex, $t > 0$

\mysubsection{Concave Functions}
$f$ concave $\Leftrightarrow$ $-f$ convex
\textbf{Examples}: $\log(x)$, $\sqrt{x}$, $x^p$ for $0 < p < 1$

\mysection{CONVEX OPTIMIZATION}

\mysubsection{Problem Classification}
\textbf{Convex Problem}: $\min f(x)$ s.t. $g_i(x) \leq b_i$, $h_j(x) = c_j$
where $f, g_i$ convex, $h_j$ linear (affine)

\textbf{Key Property}: Local minimum = Global minimum

\mysubsection{Level Sets}
$L_\alpha = \{x | f(x) \leq \alpha\}$ (sublevel set)
If $f$ convex, then $L_\alpha$ convex $\forall \alpha$

\mysubsection{Epigraph}
$\text{epi}(f) = \{(x,t) | f(x) \leq t\}$
$f$ convex $\Leftrightarrow$ $\text{epi}(f)$ convex

\mysection{LINEAR PROGRAMMING}

\mysubsection{Standard Form Conversion}
\textbf{Max to Min}: $\max c^Tx = -\min(-c^Tx)$
\textbf{Free variables}: $x = x^+ - x^-$, $x^+, x^- \geq 0$
\textbf{Inequality to equality}: Add slack: $\leq$ becomes $= $ with $s \geq 0$
\textbf{$\geq$ constraints}: Subtract surplus: $\geq$ becomes $=$ with $s \geq 0$

\mysubsection{Fundamental Theorem}
If LP has optimal solution, then $\exists$ optimal BFS
Optimal value achieved at extreme point of feasible region

\mysubsection{Basic Solutions}
Given $Ax = b$, $A \in \mathbb{R}^{m \times n}$, $m < n$:
\textbf{Basic solution}: Choose $m$ linearly independent columns (basis)
Set $n-m$ nonbasic variables to 0, solve for basic variables
\textbf{BFS}: Basic solution that is feasible ($x \geq 0$)
\textbf{Degenerate BFS}: Some basic variables = 0

\mysubsection{Extreme Points \& Rays}
\textbf{Extreme point}: Cannot be written as convex combination of other points
\textbf{Extreme ray}: Unbounded direction from feasible region
\textbf{Fundamental Theorem}: BFS $\Leftrightarrow$ Extreme point

\mysubsection{Simplex Method}
1. Start with initial BFS
2. Check optimality: $c^T - c_B^TB^{-1}A \leq 0$? (reduced costs)
3. Choose entering variable: Most negative reduced cost
4. Minimum ratio test: $\theta = \min\{x_{B_i}/y_i | y_i > 0\}$
5. Choose leaving variable: Variable achieving minimum ratio
6. Update basis and repeat

\textbf{Reduced costs}: $\bar{c}_j = c_j - c_B^TB^{-1}A_j$
\textbf{Optimality}: $\bar{c} \leq 0$ for minimization
\textbf{Unboundedness}: Some $\bar{c}_j < 0$ but all $y_j \leq 0$

\mysubsection{Degeneracy \& Cycling}
\textbf{Degeneracy}: $\theta = 0$ in ratio test
\textbf{Bland's rule}: Choose smallest index to avoid cycling

\mysection{DUALITY THEORY}

\mysubsection{Primal-Dual Pairs}
\textbf{Primal}: $\min c^Tx$ s.t. $Ax = b$, $x \geq 0$
\textbf{Dual}: $\max b^Ty$ s.t. $A^Ty \leq c$

\textbf{General form}:
\begin{tabular}{|c|c|c|}
\hline
Primal & & Dual \\
\hline
$\min c^Tx$ & & $\max b^Ty$ \\
$Ax \geq b$ & $\leftrightarrow$ & $y \geq 0$ \\
$Ax = b$ & $\leftrightarrow$ & $y$ free \\
$x \geq 0$ & $\leftrightarrow$ & $A^Ty \leq c$ \\
$x$ free & $\leftrightarrow$ & $A^Ty = c$ \\
\hline
\end{tabular}

\mysubsection{Duality Theorems}
\textbf{Weak duality}: $b^Ty \leq c^Tx$ $\forall$ feasible $x,y$
\textbf{Strong duality}: If primal has optimal solution, then dual has optimal solution with same objective value
\textbf{Complementary slackness}: At optimality:
- $x_j > 0 \Rightarrow \sum_i y_i A_{ij} = c_j$
- $y_i > 0 \Rightarrow \sum_j A_{ij}x_j = b_i$

\mysection{TRANSPORTATION \& ASSIGNMENT}

\mysubsection{Transportation Problem}
$\min \sum_{i,j} c_{ij}x_{ij}$
s.t. $\sum_j x_{ij} = s_i$ (supply), $\sum_i x_{ij} = d_j$ (demand), $x_{ij} \geq 0$

\textbf{Balanced}: $\sum s_i = \sum d_j$
\textbf{Feasibility}: Problem feasible iff $\sum s_i = \sum d_j$
\textbf{Basic variables}: $m + n - 1$ in BFS
\textbf{Degeneracy}: When $< m + n - 1$ positive variables

\mysubsection{Assignment Problem}
Special case: $s_i = d_j = 1$, $x_{ij} \in \{0,1\}$
LP relaxation gives integer solution automatically

\mysection{SENSITIVITY ANALYSIS}

\mysubsection{RHS Changes}
Current solution remains optimal for $b' = b + \Delta b$ if:
$B^{-1}(b + \Delta b) \geq 0$

\mysubsection{Cost Coefficient Changes}
Current basis remains optimal for $c' = c + \Delta c$ if:
$\bar{c} + \Delta \bar{c} \leq 0$ where $\Delta \bar{c} = \Delta c - c_B^TB^{-1}A$

\mysubsection{Shadow Prices}
Dual optimal solution $y^*$ gives shadow prices
Change in optimal value per unit increase in RHS

\mysection{NETWORK FLOWS}

\mysubsection{Max Flow Problem}
$\max f$ s.t. flow conservation at each node
Capacity constraints: $x_{ij} \leq u_{ij}$
\textbf{Max-flow min-cut theorem}: Max flow = Min cut capacity

\mysubsection{Min Cost Flow}
$\min \sum c_{ij}x_{ij}$ s.t. flow conservation, capacity constraints
Generalizes transportation and assignment problems

\mysubsection{Shortest Path}
Special case of min cost flow with unit supply/demand
Dijkstra's algorithm for non-negative weights
Bellman-Ford for general weights

\mysection{NONLINEAR PROGRAMMING}

\mysubsection{KKT Conditions}
For $\min f(x)$ s.t. $g_i(x) \leq 0$, $h_j(x) = 0$:

\textbf{Stationarity}: $\nabla f(x^*) + \sum \lambda_i \nabla g_i(x^*) + \sum \mu_j \nabla h_j(x^*) = 0$
\textbf{Primal feasibility}: $g_i(x^*) \leq 0$, $h_j(x^*) = 0$
\textbf{Dual feasibility}: $\lambda_i \geq 0$
\textbf{Complementary slackness}: $\lambda_i g_i(x^*) = 0$

\mysubsection{Constraint Qualifications}
\textbf{LICQ}: $\nabla g_i(x^*)$, $\nabla h_j(x^*)$ linearly independent
\textbf{MFCQ}: Mangasarian-Fromovitz CQ
\textbf{Slater's condition}: For convex problems

\mysubsection{Second-Order Conditions}
\textbf{SOSC}: $\nabla^2 L(x^*, \lambda^*, \mu^*)$ PD on critical cone

\mysection{LAGRANGIAN DUALITY}

\mysubsection{Lagrangian Function}
$L(x, \lambda, \mu) = f(x) + \sum \lambda_i g_i(x) + \sum \mu_j h_j(x)$

\mysubsection{Dual Function}
$q(\lambda, \mu) = \inf_x L(x, \lambda, \mu)$
Always concave in $(\lambda, \mu)$

\mysubsection{Dual Problem}
$\max q(\lambda, \mu)$ s.t. $\lambda \geq 0$
\textbf{Weak duality}: $q(\lambda, \mu) \leq p^*$ always
\textbf{Strong duality}: $q^* = p^*$ under certain conditions

\mysection{SPECIAL FUNCTIONS \& FORMULATIONS}

\mysubsection{Absolute Value Linearization}
$\min |x|$ becomes:
$\min t$ s.t. $t \geq x$, $t \geq -x$

$\min \sum |x_i|$ becomes:
$\min \sum t_i$ s.t. $t_i \geq x_i$, $t_i \geq -x_i$ $\forall i$

\mysubsection{Max/Min Linearization}
$\min \max\{f_1(x), f_2(x)\}$ becomes:
$\min t$ s.t. $t \geq f_1(x)$, $t \geq f_2(x)$

$\max \min\{f_1(x), f_2(x)\}$ becomes:
$\max t$ s.t. $t \leq f_1(x)$, $t \leq f_2(x)$

\columnbreak

\mysubsection{Piecewise Linear Functions}
$f(x) = \max\{a_1x + b_1, a_2x + b_2, \ldots\}$:
$\min f(x)$ becomes $\min t$ s.t. $t \geq a_ix + b_i$ $\forall i$

\mysubsection{SOS1 Variables}
At most one variable nonzero: $\sum x_i \leq 1$, $x_i \geq 0$

\mysection{EXAM-SPECIFIC CALCULATIONS}

\mysubsection{Matrix Rank Calculation}
Row reduce to echelon form, count nonzero rows
For $A = \begin{bmatrix} 0 & 3 & -2 & -1 \\ 4 & -9 & 1 & 3 \\ -6 & -15 & -1 & 5 \end{bmatrix}$:
Use Gaussian elimination to find rank = 2

\mysubsection{Gradient Examples}
$f(x_1,x_2) = \sqrt{x_1-1} + (x_2-1)^{1/4}$ at $(3,2)$:
$\frac{\partial f}{\partial x_1} = \frac{1}{2\sqrt{x_1-1}} = \frac{1}{2\sqrt{2}}$
$\frac{\partial f}{\partial x_2} = \frac{1}{4}(x_2-1)^{-3/4} = \frac{1}{4}$

\mysubsection{Hessian Examples}
$f(x_1,x_2) = 2(x_1-3)^2 + 6x_1x_2 + 2(x_2+3)^2 + 6x_1 + 2x_2$ at $(2,3)$:
$H = \begin{bmatrix} 4 & 6 \\ 6 & 4 \end{bmatrix}$

\mysubsection{Convexity Tests}
Check if $\nabla^2 f \succeq 0$:
For $H = \begin{bmatrix} a & b \\ b & c \end{bmatrix}$: PSD iff $a \geq 0$, $\det(H) \geq 0$

\mysubsection{Set Properties}
$X = \{x \in [-1,1]^5 : \sum x_i \geq -2\}$:
- \textbf{Bounded}: Yes (box constraints)
- \textbf{Closed}: Yes (closed inequalities)
- \textbf{Convex}: Yes (box + linear constraint)

\mysection{PROBLEM TYPES \& OUTCOMES}

\mysubsection{LP Outcomes}
1. \textbf{Optimal}: Finite optimal value
2. \textbf{Unbounded}: Objective can be made arbitrarily good
3. \textbf{Infeasible}: No point satisfies all constraints

\mysubsection{Feasibility Tests}
Transportation problem infeasible if:
$\sum s_i < \max\{d_j\}$ (total supply < max single demand)

\mysubsection{Boundedness Tests}
Set is bounded if can be contained in some ball of finite radius
Polyhedron bounded iff no extreme rays exist

\mysection{DECISION VARIABLES \& MODELING}

\mysubsection{Here-and-Now vs Wait-and-See}
\textbf{Here-and-now}: Decision before uncertainty realized
\textbf{Wait-and-see}: Decision after uncertainty realized
Wait-and-see always gives better or equal objective value

\mysubsection{Variable Types}
\textbf{Continuous}: $x \in \mathbb{R}$
\textbf{Integer}: $x \in \mathbb{Z}$
\textbf{Binary}: $x \in \{0,1\}$
\textbf{Semi-continuous}: $x = 0$ or $x \in [l,u]$

\mysection{KEY THEOREMS \& RESULTS}

\mysubsection{Fundamental Theorem of LP}
If LP has optimal solution, then has optimal BFS
Every bounded polyhedron has finite number of extreme points

\mysubsection{Caratheodory's Theorem}
Any point in convex hull of $S$ can be expressed as convex combination of at most $d+1$ points from $S$ (where $d$ is dimension)

\mysubsection{Farkas' Lemma}
Either $Ax = b$, $x \geq 0$ has solution, or $A^Ty \geq 0$, $b^Ty < 0$ has solution, but not both

\mysection{COMPUTATIONAL NOTES}

\mysubsection{Complexity}
\textbf{LP}: Polynomial time (interior point methods)
\textbf{Simplex}: Exponential worst case, polynomial average
\textbf{Integer programming}: NP-hard
\textbf{Convex optimization}: Polynomial time

\mysubsection{Common Mistakes}
- Confusing convex function with concave function
- Wrong constraint qualification application  
- Incorrect dual formulation
- Matrix definiteness errors
- Sign errors in KKT conditions
