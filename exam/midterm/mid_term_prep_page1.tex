\mysection{OPTIMIZATION FUNDAMENTALS}

\mysubsection{Standard Forms}
\textbf{General}: $\min f(x)$ s.t. $g_i(x) \leq b_i$, $h_j(x) = c_j$, $x \in X$
\textbf{LP Standard}: $\min c^Tx$ s.t. $Ax = b$, $x \geq 0$

\mysubsection{Linear Algebra (Past Exam Focus)}
\textbf{Matrix rank}: Count linearly independent rows/columns
\textbf{Example}: $A = \begin{bmatrix} 0 & 3 & -2 & -1 \\ 4 & -9 & 1 & 3 \\ -6 & -15 & -1 & 5 \end{bmatrix}$ has rank = 2

\textbf{System $Ax = b$}:
- $m = n$, full rank: unique solution
- $m < n$: infinite solutions  
- $m > n$: over-determined
- Inconsistent: $\text{rank}(A) < \text{rank}([A|b])$

\mysubsection{Gradients \& Hessians (Past Exam)}
$\nabla f = [\frac{\partial f}{\partial x_1}, \ldots, \frac{\partial f}{\partial x_n}]^T$
$\nabla^2 f = [\frac{\partial^2 f}{\partial x_i \partial x_j}]$

\textbf{Example 1}: $f(x_1,x_2) = \sqrt{x_1-1} + (x_2-1)^{1/4}$ at $(3,2)$
$\nabla f = \begin{bmatrix} \frac{1}{2\sqrt{2}} \\ \frac{1}{4} \end{bmatrix}$

\textbf{Example 2}: $f(x_1,x_2) = 2(x_1-3)^2 + 6x_1x_2 + 2(x_2+3)^2 + 6x_1 + 2x_2$ at $(2,3)$
$H = \begin{bmatrix} 4 & 6 \\ 6 & 4 \end{bmatrix}$

\mysubsection{Taylor Expansions}
\textbf{1st order}: $f(x) \approx f(x_0) + \nabla f(x_0)^T(x-x_0)$
\textbf{Example}: $f(x,y) = x^3 - 2xy + y^2$ at $(-2,3)$ gives $13x - 3y - 6$

\mysubsection{Optimality Conditions}
\textbf{1st order necessary}: $\nabla f(x^*) = 0$
\textbf{2nd order (at stationary $x^*$)}:
- $\nabla^2 f(x^*) \succ 0$ (positive definite) $\Rightarrow$ strict local min
- $\nabla^2 f(x^*) \succeq 0$ but not PD (positive semidefinite) $\Rightarrow$ inconclusive (flat directions; possible min)
- $\nabla^2 f(x^*) \prec 0$ (negative definite) $\Rightarrow$ strict local max
- $\nabla^2 f(x^*) \preceq 0$ but not ND (negative semidefinite) $\Rightarrow$ inconclusive (flat directions; possible max)
- $\nabla^2 f(x^*)$ indefinite $\Rightarrow$ saddle (neither min nor max)

\textbf{Matrix definiteness test}:
For $\begin{bmatrix} a & b \\ b & c \end{bmatrix}$: PD if $a > 0, \det > 0$; PSD if $a \geq 0, \det \geq 0$

\mysection{CONVEXITY (Major Exam Topic)}

\mysubsection{Convex Sets}
\textbf{Definition}: $\lambda x + (1-\lambda)y \in S$ $\forall x,y \in S$, $\lambda \in [0,1]$

\textbf{Examples from past exams}:
- $\{x \in \mathbb{R}^2 | \min\{x_1,x_2\} \geq 1\}$: NOT convex
- $\{x \in \mathbb{R}^2 | x_1 \geq \ln x_2, x_2 > 0\}$: NOT convex  
- $\{x \in \mathbb{R}^2 | x_1^2 + x_2^2 \geq 4\}$: NOT convex
- Hyperplane $\{x | a^Tx = b\}$: convex
- Halfspace $\{x | a^Tx \leq b\}$: convex

\mysubsection{Set Properties (Past Exam)}
$X = \{x \in [-1,1]^5 : \sum_{i=1}^5 x_i \geq -2\}$:
- \textbf{Bounded}: Yes (box constraints)
- \textbf{Closed}: Yes (closed inequality)

\mysubsection{Convex Functions}
\textbf{Definition}: $f(\lambda x + (1-\lambda)y) \leq \lambda f(x) + (1-\lambda)f(y)$

\textbf{Tests}:
1. \textbf{2nd order}: $\nabla^2 f(x) \succeq 0$ (PSD)
2. \textbf{1st order}: $f(y) \geq f(x) + \nabla f(x)^T(y-x)$

\textbf{Examples from exams}:
- $f(x_1,x_2) = 2|x_1+1| + |x_2-x_1|$: convex
- $f(x) = -2x^2 + 6x + 10$: NOT convex (concave)
- $f(x) = x^5$ on $[-1,1]$: NOT convex
- $f(x) = \ln(x^2)$ on $(0,\infty)$: convex

\mysection{CONVEX OPTIMIZATION PROBLEMS}

\mysubsection{Problem Classification (Past Exam)}
\textbf{Convex problem}: $\min f(x)$ s.t. $g_i(x) \leq b_i$, $h_j(x) = c_j$
where $f, g_i$ convex, $h_j$ linear

\textbf{Examples from past exams}:
- $\min\{x^4 + y^4 : x^2 + y^2 \geq 20\}$: NOT convex (feasible region not convex)
- $\max\{\cos(x) : x \in [0,2\pi]\}$: NOT convex ($\cos$ not concave on interval)
- $\min\{\ln(x) : x \geq 1\}$: convex
- $\min\{-x - 8y : x + 7y \geq 10, e^x \leq 1\}$: convex

\mysection{LINEAR PROGRAMMING}

\mysubsection{Standard Form Conversion}
\textbf{From}: $\max\{2x_1 - 3x_2 : x_1 + x_2 \leq 2, x_1 + 2x_2 \geq 3\}$
\textbf{To}: $\min\{-2x_1^+ + 2x_1^- + 3x_2^+ - 3x_2^- : x_1^+ - x_1^- + x_2^+ - x_2^- + s_1 = 2, x_1^+ - x_1^- + 2x_2^+ - 2x_2^- - s_2 = 3, \text{all} \geq 0\}$

\mysubsection{Basic Solutions \& Extreme Points}
\textbf{Basic feasible solution (BFS)}: Extreme point of feasible region
\textbf{Degenerate BFS}: Some basic variables = 0
\textbf{Optimality}: If LP has optimal solution, then has optimal BFS

\mysubsection{Simplex Method}
\textbf{Steps}: Start BFS → Check optimality → Choose entering variable → Ratio test → Update basis
\textbf{Reduced costs}: For min problem, optimal if all $\bar{c}_j \geq 0$
\textbf{Degeneracy example}: Reduced costs $(3,-1,0)$ with ratios $(1,0)$ → degenerate

\mysection{TRANSPORTATION PROBLEM}

\mysubsection{Standard Form}
$\min \sum_{i=1}^m \sum_{j=1}^n c_{ij}x_{ij}$
s.t. $\sum_{j=1}^n x_{ij} \leq s_i$ (supply), $\sum_{i=1}^m x_{ij} \geq d_j$ (demand), $x_{ij} \geq 0$

\textbf{Infeasibility condition}: Total supply $\sum s_i < \max\{d_j\}$

\mysection{SPECIAL FORMULATIONS}

\mysubsection{Absolute Value Problems (Past Exam)}
$\min |x-2| + 3|1-x|$ s.t. $x \geq 0$
$\rightarrow \min z_1 + 3z_2$ s.t. $z_1 \geq x-2, z_1 \geq -(x-2), z_2 \geq 1-x, z_2 \geq -(1-x), x \geq 0$

\mysubsection{Max/Min Functions (Past Exam)}
$\min\{\frac{1}{2}(|x-4| + |x+5|) + \frac{1}{2}\max\{|x-4|,|x+5|\}\}$
$\rightarrow \min \frac{1}{2}z_1 + \frac{1}{2}z_2$ s.t. $z_1 \geq |x-4| + |x+5|, z_2 \geq |x-4|, z_2 \geq |x+5|$

\mysection{SPECIFIC CALCULATIONS}

\mysubsection{Feasible Point Tests (Past Exam)}
$X = \{x \in \mathbb{R}^2 | x_1 - x_2 \leq -1, 2x_1 - x_2 \leq 0, x_2 \geq -\frac{1}{2}\}$
Test $(0,-\frac{1}{2})$: $0-(-\frac{1}{2}) = \frac{1}{2} \not\leq -1$ → NOT feasible

\mysubsection{Optimization Outcomes (Past Exam)}
$\max\{412x - 511y : 0 \leq x \leq 1, y \leq 1\}$
Since coefficients positive for $x$, negative for $y$: optimal at $x=1, y=0$

\mysubsection{Piecewise Linear Functions}
$f(x) = \min\{10x-1, 2x+10, 20\}$: Can be reformulated as LP
$f(x) = \sum_{i=1}^{2019} 2^{-i}|e^{i/2}x - \sin(i)|$: NOT standard LP form

\mysection{DECISION CONCEPTS}

\mysubsection{Timing Decisions}
\textbf{Here-and-now}: Decision before uncertainty realized
\textbf{Wait-and-see}: Decision after uncertainty realized

\mysubsection{Function Properties}
\textbf{Level sets}: Convex function does NOT always have convex level sets
\textbf{Boundary optimization}: Max of convex function over compact set achieved at boundary
\textbf{Uniqueness}: Non-convex problems can have multiple optimal solutions

\mysection{KEY ABBREVIATIONS}
LP=Linear Programming, BFS=Basic Feasible Solution, PD=Positive Definite, PSD=Positive Semidefinite
